```csharp
// 文件: AutoBattleAI.cs
public void StartBattleSequence()
{
    Debug.Log("🎯 ========== AutoBattleAI.StartBattleSequence 被调用 ==========");

    if (isBattleActive)
    {
        Debug.LogWarning("战斗已经在进行中！");
        return;
    }

    // 步骤1: 收集所有参战角色
    var allCombatants = new List<CharacterStats>();
    CharacterStats[] allCharacters = FindObjectsOfType<CharacterStats>();
    
    foreach (var character in allCharacters)
    {
        // 包括所有生命值>0 的角色或处于昏迷状态的角色
        if (character.CurrentHitPoints > 0 || 
            character.HasStatusEffect(StatusEffectType.Unconscious))
        {
            allCombatants.Add(character);
            Debug.Log($"✓ 添加参战角色: {character.GetDisplayName()} " +
                     $"(阵营: {character.battleSide}, 生命值: {character.CurrentHitPoints})");
        }
    }

    if (allCombatants.Count < 2)
    {
        Debug.LogWarning($"参战角色不足 (仅有{allCombatants.Count}人)，无法开始战斗");
        return;
    }

    // 步骤2: 执行先攻检定 ✨ 关键！
    Debug.Log("执行先攻检定...");
    initiativeOrder = HorizontalCombatRules.RollAndSortInitiative(allCombatants);

    // 步骤3: 初始化回合控制变量
    currentTurnIndex = 0;
    isBattleActive = true;
    isProcessingTurn = false;
    turnTimer = 0f;

    // 输出先攻顺序供调试
    Debug.Log("【先攻顺序】");
    for (int i = 0; i < initiativeOrder.Count; i++)
    {
        var entry = initiativeOrder[i];
        Debug.Log($"  {i + 1}. {entry.character.GetDisplayName()} " +
                 $"(掷骰: {entry.initiativeRoll}, 先攻值: {entry.initiativeValue})");
    }

    Debug.Log("战斗开始！进入主循环");
    StartNextTurn();
}
```

```csharp
// 文件: HorizontalCombatRules.cs
public static List<InitiativeEntry> RollAndSortInitiative(List<CharacterStats> combatants)
{
    var list = new List<InitiativeEntry>();
    if (combatants == null || combatants.Count == 0)
        return list;

    // 遍历每个参战者
    foreach (var c in combatants)
    {
        if (c == null) continue;

        // 获取敏捷修正值
        var snap = TryGetSnapshot(c);
        int dexMod = snap.HasValue 
            ? snap.Value.DexMod 
            : GetAbilityModifier(c.dexterity);

        // 掷骰: d20 + 敏捷修正
        int roll = UnityEngine.Random.Range(1, 21);
        int total = roll + dexMod;

        // 创建先攻记录条目
        var entry = new InitiativeEntry(c, total);
        list.Add(entry);

        Debug.Log($"🎲 {c.GetDisplayName()}: d20={roll} + DEX修正={dexMod} = {total}");
        try 
        { 
            GameLog.LogInitiative(c.GetDisplayName(), roll, dexMod, total); 
        }
        catch (System.Exception ex) 
        { 
            Debug.LogWarning($"记录先攻日志失败: {ex.Message}"); 
        }
    }

    // 按先攻值降序排列 (高优先)
    var sortedList = list.OrderByDescending(e => e.initiativeValue).ToList();
    
    return sortedList;
}

// InitiativeEntry 定义
[System.Serializable]
public class InitiativeEntry
{
    public CharacterStats character;
    public int initiativeRoll;      // 掷骰结果 (1-20)
    public int initiativeValue;     // 最终先攻值 (roll + modifier)
    
    public InitiativeEntry(CharacterStats c, int value)
    {
        character = c;
        initiativeValue = value;
    }

    public bool CanAct() => character != null && character.CurrentHitPoints > 0;
    
    public void MarkAsActed() => character?.CompleteAction();
}
```

### 1.3 回合处理核心循环

```csharp
// 文件: AutoBattleAI.cs
private void Update()
{
    if (!isBattleActive || !enableAutoBattle)
        return;

    // 定时器驱动的回合处理（避免每帧处理）
    if (!isProcessingTurn)
    {
        turnTimer += Time.deltaTime;
        
        if (turnTimer >= decisionDelay)  // 通常 1.0 秒
        {
            ProcessCurrentTurn();  // ← 处理当前角色的回合
        }
    }
}

private void ProcessCurrentTurn()
{
    if (isProcessingTurn)
        return;

    // 获取当前行动的角色
    var currentEntry = GetCurrentInitiativeEntry();
    if (currentEntry == null)
    {
        AdvanceToNextTurn();
        return;
    }

    var character = currentEntry.character;

    // 检查战斗是否已结束
    if (IsBattleOver())
    {
        EndBattle();
        return;
    }

    // 检查角色是否可以行动
    if (!currentEntry.CanAct())
    {
        Debug.Log($"⊘ {character.GetDisplayName()} 无法行动 (生命值: {character.CurrentHitPoints})");
        AdvanceToNextTurn();
        return;
    }

    // 特殊处理: 昏迷状态需要进行死亡豁免
    if (character.HasStatusEffect(StatusEffectType.Unconscious))
    {
        Debug.Log($"💤 {character.GetDisplayName()} 处于昏迷，执行死亡豁免...");
        character.PerformDeathSaveTick();  // 进行DC10豁免投掷
        currentEntry.MarkAsActed();
        AdvanceToNextTurn();
        return;
    }

    isProcessingTurn = true;

    if (showAIThoughts)
    {
        Debug.Log($"=== {character.GetDisplayName()} 的回合开始 ===");
    }

    // AI决策: 选择最佳行动
    var chosenAction = DecideBestAction(character);
    
    if (chosenAction != null)
    {
        // 执行选定的行动并传入完成回调
        ExecuteBattleActionEvent(
            character,
            chosenAction,
            onComplete: () => {
                currentEntry.MarkAsActed();
                AdvanceToNextTurn();
            }
        );
    }
    else
    {
        // AI无法选择行动，跳过此回合
        currentEntry.MarkAsActed();
        AdvanceToNextTurn();
    }
}

private BattleAction DecideBestAction(CharacterStats character)
{
    // 简化版本：选择敌方血量最低的目标
    CharacterStats bestTarget = null;
    int lowestHP = int.MaxValue;

    CharacterStats[] allCharacters = FindObjectsOfType<CharacterStats>();
    foreach (var other in allCharacters)
    {
        // 选择对方阵营的活着的角色
        if (other.battleSide != character.battleSide && 
            !other.IsDownOrDead())
        {
            if (other.CurrentHitPoints < lowestHP)
            {
                lowestHP = other.CurrentHitPoints;
                bestTarget = other;
            }
        }
    }

    if (bestTarget != null)
    {
        return new BattleAction { target = bestTarget };
    }

    return null;
}

private void AdvanceToNextTurn()
{
    isProcessingTurn = false;
    turnTimer = 0f;
    currentTurnIndex++;

    if (currentTurnIndex >= initiativeOrder.Count)
    {
        currentTurnIndex = 0;  // 循环回第一个
    }

    StartNextTurn();
}

private InitiativeEntry GetCurrentInitiativeEntry()
{
    if (initiativeOrder == null || initiativeOrder.Count == 0)
        return null;

    if (currentTurnIndex < 0 || currentTurnIndex >= initiativeOrder.Count)
        return null;

    return initiativeOrder[currentTurnIndex];
}

private bool IsBattleOver()
{
    // 检查是否一方全��阵亡
    bool playerAlive = false, enemyAlive = false;

    foreach (var entry in initiativeOrder)
    {
        if (entry.character.battleSide == BattleSide.Player && 
            entry.character.CurrentHitPoints > 0)
            playerAlive = true;

        if (entry.character.battleSide == BattleSide.Enemy && 
            entry.character.CurrentHitPoints > 0)
            enemyAlive = true;
    }

    return !playerAlive || !enemyAlive;
}
```

```csharp
// 文件: DND_CharacterAdapter.cs
public void ExecuteMeleeAttack(
    Transform target,
    System.Action onAttackHit = null,
    System.Action onComplete = null)
{
    // 【预检查】
    if (target == null)
    {
        Debug.LogError($"{gameObject.name}: 目标为空！");
        return;
    }

    if (IsTerminalState())
    {
        Debug.LogWarning($"{gameObject.name}: 已死亡/昏迷，无法攻击");
        return;
    }

    if (isAnimating)
    {
        Debug.LogWarning($"{gameObject.name}: 正在动画中，无法开始新攻击");
        return;
    }

    isAnimating = true;
    Debug.Log($"[近战攻击开始] {gameObject.name} → {target.name}");

    // 【阶段1: 位移】
    Vector3 attackPosition = CalculateAttackPosition(target);
    float moveDistance = Vector3.Distance(transform.position, attackPosition);
    float moveDuration = moveDistance / animationConfig.moveSpeed;

    Debug.Log($"  移动距离: {moveDistance:F2}, 预计时长: {moveDuration:F2}s");

    PlayWalkAnimation();

    // Kill 旧Tween确保没有冲突
    if (currentMoveTween != null && currentMoveTween.IsActive())
    {
        currentMoveTween.Kill();
        currentMoveTween = null;
    }

    const float arriveSnapThreshold = 0.1f;
    bool arrivalTriggered = false;

    // 预定义抵达处理逻辑，供OnUpdate和OnComplete共用
    System.Action proceedAfterArrive = () =>
    {
        if (arrivalTriggered) return;
        arrivalTriggered = true;

        if (IsTerminalState())
        {
            Debug.Log($"{gameObject.name}: 移动途中角色已死亡，中止攻击");
            isAnimating = false;
            return;
        }

        Debug.Log($"[阶段切换] {gameObject.name} 已抵达攻击位置");

        // 【阶段2: 攻击】
        isForceAttackAnimation = true;  // 上锁，防止walk覆盖
        CutImmediatelyForAttack();      // 清轨无缝切换

        ExecuteAttackAtPosition(
            target,
            onAttackHit,
            onComplete: () => {
                Debug.Log($"[阶段3] {gameObject.name} 攻击完成，返回原位");

                if (IsTerminalState())
                {
                    isAnimating = false;
                    return;
                }

                // 返回原位
                PlayWalkAnimation();
                currentMoveTween = transform.DOMove(originalPosition, moveDuration)
                    .SetEase(animationConfig.moveEase)
                    .OnComplete(() => {
                        Debug.Log($"[完成] {gameObject.name} 已返回原位");
                        isAnimating = false;
                        if (!IsTerminalState())
                            PlayIdleAnimation();
                        onComplete?.Invoke();
                    });
            }
        );
    };

    // 【执行位移Tween】
    currentMoveTween = transform.DOMove(attackPosition, moveDuration)
        .SetEase(animationConfig.moveEase)
        .OnUpdate(() => {
            // 实时检查是否已接近目标
            float distanceToTarget = Vector3.Distance(transform.position, attackPosition);
            
            if (distanceToTarget <= arriveSnapThreshold)
            {
                // 精准对齐到目标位置
                transform.position = attackPosition;
                
                if (currentMoveTween != null && currentMoveTween.IsActive())
                {
                    currentMoveTween.Kill();
                    currentMoveTween = null;
                }

                proceedAfterArrive();
            }
        })
        .OnComplete(() => {
            // 正常完成时的回调
            if (!arrivalTriggered)
            {
                proceedAfterArrive();
            }
        });
}

private Vector3 CalculateAttackPosition(Transform target)
{
    // 计算在目标前方 attackDistance 距离处的位置
    Vector3 directionToTarget = (target.position - transform.position).normalized;
    Vector3 attackPos = target.position - directionToTarget * animationConfig.attackDistance;
    
    // 限制在同一高度（Y轴）
    attackPos.y = transform.position.y;
    
    return attackPos;
}

private void CutImmediatelyForAttack()
{
    if (skeletonAnimation?.AnimationState == null)
        return;

    try
    {
        // 1. 清除所有轨道（包括walk的残留）
        skeletonAnimation.AnimationState.ClearTracks();
        
        // 2. 复位骨架到初始姿态
        skeletonAnimation.Skeleton.SetToSetupPose();
        
        // 3. 立即应用更新（不等下一帧）
        skeletonAnimation.AnimationState.Apply(skeletonAnimation.Skeleton);
        skeletonAnimation.Skeleton.UpdateWorldTransform();
        
        // 4. 额外刷新确保渲染更新
        try 
        { 
            skeletonAnimation.Update(0f); 
        }
        catch (System.Exception ex) 
        { 
            Debug.LogWarning($"Update(0) 失败: {ex.Message}"); 
        }

        Debug.Log($"{gameObject.name}: 清轨完成，准备播放攻击动画");
    }
    catch (System.Exception ex)
    {
        Debug.LogError($"{gameObject.name}: CutImmediatelyForAttack 失败: {ex.Message}");
    }
}
```

```csharp
// 文件: DND_CharacterAdapter.cs
private void ExecuteAttackAtPosition(
    Transform target,
    System.Action onAttackHit = null,
    System.Action onComplete = null)
{
    Debug.Log($"【攻击执行】 {gameObject.name} 在当前位置执行攻击");

    isForceAttackAnimation = true;
    CutImmediatelyForAttack();

    // 播放攻击动画
    PlayAttackAnimation();

    // 获取攻击动画时长
    string attackAnimName = FindBestAttackAnimationName();
    float animDuration = GetAnimationDuration(attackAnimName);
    if (animDuration <= 0) animDuration = 1.0f;

    Debug.Log($"  攻击动画: '{attackAnimName}', 时长: {animDuration:F2}s");

    // 设置命中回调标志
    bool hasHitTriggered = false;
    bool hasCompleteTriggered = false;

    // 【方案1】 Spine事件触发（主路径）
    System.Action hitCallback = () => {
        if (hasHitTriggered) return;
        hasHitTriggered = true;
        Debug.Log($"💥 {gameObject.name} 攻击命中事件触发（Spine事件）");
        onAttackHit?.Invoke();
    };
    OnAttackHit += hitCallback;

    // 【方案2】 备用触发器（防止Spine事件丢失）
    DOVirtual.DelayedCall(animDuration * 0.5f, () => {
        if (!hasHitTriggered && !IsTerminalState())
        {
            hasHitTriggered = true;
            Debug.Log($"💥 {gameObject.name} 攻击命中事件触发（备用计时器）");
            onAttackHit?.Invoke();
        }
    });

    // 【方案1】 Spine动画完成事件（主路径）
    System.Action completeCallback = () => {
        if (hasCompleteTriggered) return;
        hasCompleteTriggered = true;
        Debug.Log($"✓ {gameObject.name} 攻击完成（Spine事件）");
        
        isForceAttackAnimation = false;  // 解除锁
        onComplete?.Invoke();
        
        // 清理事件监听
        OnAttackHit -= hitCallback;
        OnAnimationComplete -= completeCallback;
    };
    OnAnimationComplete += completeCallback;

    // 【方案2】 备用完成触发
    DOVirtual.DelayedCall(animDuration + 0.1f, () => {
        if (!hasCompleteTriggered)
        {
            hasCompleteTriggered = true;
            Debug.Log($"✓ {gameObject.name} 攻击完成（备用计时器）");
            
            isForceAttackAnimation = false;
            onComplete?.Invoke();
            
            OnAttackHit -= hitCallback;
            OnAnimationComplete -= completeCallback;
        }
    });
}

// 文件: AutoBattleAI.cs
private void ProcessAttackHit(
    CharacterStats attacker,
    CharacterStats target,
    bool isMeleeAttack)
{
    if (attacker == null || target == null) return;

    Debug.Log($"【伤害计算】 {attacker.GetDisplayName()} → {target.GetDisplayName()}");

    // 步骤1: 检查特殊状态（如昏迷）
    int advantageFlag = 0;
    if (target.HasStatusEffect(StatusEffectType.Unconscious))
    {
        // 近战时有优势（d20取最大），远程时有劣势（d20取最小）
        advantageFlag = isMeleeAttack ? 1 : -1;
        Debug.Log($"  目标处于昏迷: 攻击掷骰标志 = {advantageFlag}");
    }

    // 步骤2: 调用战斗规则进行完整结算
    var attackResult = HorizontalCombatRules.ResolveAttack(
        attacker,
        target,
        advantageFlag,
        isMeleeAttack
    );

    // 步骤3: 如果命中，应用伤害
    if (attackResult.isHit)
    {
        int damage = attackResult.damage;
        bool isCritical = attackResult.isCritical;

        string critText = isCritical ? " 【暴击】" : "";
        Debug.Log($"  ✓ 命中! 伤害: {damage}{critText}");

        // 目标接收伤害
        target.ReceiveDamage(damage, isCritical);

        // 检查目标是否阵亡
        if (target.IsDownOrDead())
        {
            Debug.Log($"  ☠️ {target.GetDisplayName()} 已阵亡！");
        }
    }
    else
    {
        Debug.Log($"  ✗ 未命中");
    }
}

// 文件: HorizontalCombatRules.cs
public static AttackResult ResolveAttack(
    CharacterStats attacker,
    CharacterStats target,
    int advantageFlag = 0,
    bool isMeleeAttack = true)
{
    var result = new AttackResult { isHit = false, damage = 0 };

    if (attacker == null || target == null)
        return result;

    // 【步骤1】 确定攻击类型与武器
    bool isSpell = attacker.template?.defaultAttackType == DefaultAttackType.Spell;
    var weapon = isSpell ? null : FindEquippedWeapon(attacker);

    // 【步骤2】 计算攻击加值
    string hitAbility;
    string attackName;
    int attackBonus = GetAttackBonus(attacker, isMeleeAttack, out hitAbility, out attackName, weapon);

    // 【步骤3】 掷骰: d20 + 加值
    int roll1 = UnityEngine.Random.Range(1, 21);
    int roll2 = UnityEngine.Random.Range(1, 21);
    int d20 = (advantageFlag > 0)  ? Mathf.Max(roll1, roll2)     // 优势
            : (advantageFlag < 0)  ? Mathf.Min(roll1, roll2)     // 劣势
            : roll1;                                               // 正常

    int totalAttack = d20 + attackBonus;

    // 【步骤4】 暴击判定
    result.isCritical = (d20 == 20);

    // 【步骤5】 命中判定 vs 目标AC
    var targetSnapshot = TryGetSnapshot(target);
    int targetAC = targetSnapshot.HasValue ? targetSnapshot.Value.armorClass : target.CurrentArmorClass;
    
    result.isHit = result.isCritical || totalAttack >= targetAC;

    Debug.Log($"  掷骰: d20({roll1},{roll2}) → {d20}, " +
             $"加值: {attackBonus}, 总计: {totalAttack} vs AC {targetAC} → {(result.isHit ? "命中" : "未命中")}");

    // 【步骤6】 伤害计算
    if (result.isHit)
    {
        int diceSize, rolled, dmgMod;
        string dmgAbility;
        result.damage = CalculateDamageUnified(
            attacker,
            result.isCritical,
            isSpell,
            hitAbility,
            out diceSize,
            out rolled,
            out dmgMod,
            weapon,
            out dmgAbility
        );

        Debug.Log($"  伤害计算: {rolled}d{diceSize} = {rolled}, " +
                 $"{dmgAbility}修正: {dmgMod}, 总伤害: {result.damage}");
    }

    // 【步骤7】 发布伤害事件（事件总线）
    if (result.isHit)
    {
        try
        {
            var channel = GetDamageEventChannel(attacker, target);
            if (channel != null)
            {
                var damageInfo = new DamageInfo(target, attacker, result.damage, result.isCritical);
                channel.RaiseEvent(damageInfo);
                Debug.Log($"  [事件] DamageEventChannel.RaiseEvent() 已发布");
            }
        }
        catch (System.Exception ex)
        {
            Debug.LogWarning($"发布伤害事件异常: {ex.Message}");
        }
    }

    return result;
}
```

### 2.3 受伤处理与死亡流程

```csharp
// 文件: CharacterStats.cs
public void ReceiveDamage(int damageAmount, bool isCritical = false)
{
    if (damageAmount < 0) damageAmount = 0;

    int oldHP = currentHitPoints;
    currentHitPoints -= damageAmount;
    if (currentHitPoints < 0) currentHitPoints = 0;

    Debug.Log($"【伤害处理】 {GetDisplayName()}: {oldHP} → {currentHitPoints} " +
             $"(-{damageAmount}){(isCritical ? " 【暴击】" : "")}");

    // 【步骤1】 发布伤害事件（通知UI更新）
    try
    {
        if (damageEventChannel != null)
        {
            var info = new DamageInfo(this, null, damageAmount, isCritical);
            damageEventChannel.RaiseEvent(info);
        }
    }
    catch (System.Exception ex)
    {
        Debug.LogWarning($"发布伤害事件失败: {ex.Message}");
    }

    // 【步骤2】 播放受击动画
    var adapter = GetComponent<DND_CharacterAdapter>();
    if (adapter != null && !IsDownOrDead())
    {
        adapter.PlayHitAnimation();
    }

    // 【步骤3】 检查是否阵亡
    if (currentHitPoints <= 0)
    {
        BecomeUnconsciousOrDead();
    }
}

public void BecomeUnconsciousOrDead()
{
    if (statusEffects.Contains(StatusEffectType.Unconscious))
        return;  // 已经处于昏迷状态

    Debug.Log($"【阵亡判定】 {GetDisplayName()} 生命值 <= 0，进入昏迷状态");

    // 【步骤1】 添加昏迷状态
    statusEffects.Add(StatusEffectType.Unconscious);

    // 【步骤2】 播放死亡/昏迷动画
    var adapter = GetComponent<DND_CharacterAdapter>();
    if (adapter != null)
    {
        // 优先播放死亡动画，若角色完全死亡则标记为已死亡
        adapter.PlayDeathAnimation();
        Debug.Log($"  播放死亡动画");
    }

    // 【步骤3】 标记已播放死亡动画
    HasPlayedDeath = true;

    // 【步骤4】 初始化死亡豁免计数（可选）
    unconsciousSuccessCount = 0;
    unconsciousFailureCount = 0;

    // 【步骤5】 通知战斗系统（可选）
    try
    {
        GameLog.LogAction(GetDisplayName(), "已阵亡");
    }
    catch (System.Exception ex)
    {
        Debug.LogWarning($"记录日志失败: {ex.Message}");
    }
}

public void PerformDeathSaveTick()
{
    // 每回合进行一次DC10的体质豁免投掷
    // 成功3次恢复1点HP，失败3次死亡，成功+失败同时达到3次则稳定
    
    int roll = UnityEngine.Random.Range(1, 21);
    int conMod = ConMod;
    int total = roll + conMod;

    Debug.Log($"【死亡豁免】 {GetDisplayName()}: d20={roll} + CON={conMod} = {total}");

    bool isSuccess = (total >= 10);

    if (isSuccess)
    {
        unconsciousSuccessCount++;
        Debug.Log($"  成功 (成功次数: {unconsciousSuccessCount}/3)");

        if (unconsciousSuccessCount >= 3)
        {
            // 三次成功，恢复1点HP
            currentHitPoints = 1;
            statusEffects.Remove(StatusEffectType.Unconscious);
            Debug.Log($"  ✓ 成功3次，恢复1点HP并恢复意识");
        }
    }
    else
    {
        unconsciousFailureCount++;
        Debug.Log($"  失败 (失败次数: {unconsciousFailureCount}/3)");

        if (unconsciousFailureCount >= 3)
        {
            // 三次失败，彻底死亡
            currentHitPoints = -10;
            Debug.Log($"  ☠️ 失败3次，角色已死亡");
        }
    }
}
```

```csharp
    private Spine.AnimationState cachedAnimState;

// 【优化3】 Object pooling
public class CharacterAnimationPool : MonoBehaviour
{
    [SerializeField] private GameObject characterPrefab;
    private Queue<GameObject> pool = new Queue<GameObject>();
    
    public GameObject GetCharacter()
    {
        if (pool.Count > 0)
            return pool.Dequeue();
        
        return Instantiate(characterPrefab);
    }
    
    public void ReturnCharacter(GameObject character)
    {
        character.SetActive(false);
        pool.Enqueue(character);
    }
}
```

## 5. 常见问题解决方案

### 5.1 动画不播放

**症状**: PlayAttackAnimation() 被调用但角色仍保持idle状态

**检查步骤**:
```csharp
// 1. 检查 animationConfig 是否为null
if (animationConfig == null)

    Debug.LogError($"{gameObject.name}: animationConfig 未配置！");
    return;

string attackAnim = animationConfig.attackAnimation;
Debug.Log($"寻找动画: '{attackAnim}'");

// 3. 列出Spine中所有可用动画
if (skeletonAnimation?.Skeleton?.Data != null)
{
    Debug.Log("Spine中的动画:");
    for (int i = 0; i < skeletonAnimation.Skeleton.Data.Animations.Count; i++)
    {
        string animName = skeletonAnimation.Skeleton.Data.Animations.Items[i].Name;
        Debug.Log($"  [{i}] {animName}");
        
        // 检查名称是否匹配（区分大小写）
        if (animName == attackAnim)
            Debug.Log($"    ✓ 匹配!");
    }
}

// 4. 尝试手动设置动画
try
{
    skeletonAnimation.AnimationState.SetAnimation(0, attackAnim, false);
```

- ❌ 动画名大小写不匹配 (e.g., "Attack" vs "attack")

- ❌ SkeletonAnimation 组件未正确导入
